```powershell
$ErrorActionPreference = 'Stop'
node scripts/verify-shared-catalog.mjs logs/shared-catalog.json
if ($LASTEXITCODE -ne 0) { throw 'Catalog verification failed; no approval created.' }
$proofPath = (Resolve-Path 'logs/shared-schema-verification.json').Path
$proof = Get-Content -LiteralPath $proofPath -Raw | ConvertFrom-Json -DateKind String
if ($proof.projectRef -ne 'usvopldwiwzlvoqawvcu' -or !$proof.catalogVerified -or !$proof.compatible) {
    throw 'Wrong project or incompatible evidence.'
}
$expiry = [DateTimeOffset]::UtcNow.AddMinutes(30)
$evidenceExpiry = [DateTimeOffset]::Parse($proof.checkedAt).AddHours(24)
if ($expiry -gt $evidenceExpiry) { $expiry = $evidenceExpiry }
if ($expiry -le [DateTimeOffset]::UtcNow.AddMinutes(10)) { throw 'Evidence expires too soon.' }
$runId = [guid]::NewGuid().ToString()
$approval = [ordered]@{
    projectRef = 'usvopldwiwzlvoqawvcu'
    acknowledgement = 'I approve temporary test records in shared project usvopldwiwzlvoqawvcu.'
    scope = 'rls'
    runId = $runId
    maxUsers = 2
    maxAiOperations = 0
    expiresAt = $expiry.ToString('o')
    schemaEvidenceFile = $proofPath
    schemaEvidenceSha256 = (Get-FileHash -LiteralPath $proofPath -Algorithm SHA256).Hash.ToLowerInvariant()
    appOrigin = 'http://127.0.0.1:4173'
}
$path = Join-Path (Get-Location) "logs/owner-rls-$runId.approval.json"
if (Test-Path -LiteralPath $path) { throw 'Refusing to overwrite existing approval.' }
[IO.File]::WriteAllText($path, ($approval | ConvertTo-Json), [Text.UTF8Encoding]::new($false))
Write-Output "Owner approval created: $path"
```
